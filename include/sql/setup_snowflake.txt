-- =================
-- SNOWFLAKE SETUP
-- =================

USE ROLE ACCOUNTADMIN;

-- =================
-- SNOWFLAKE SETUP:
-- Create roles, warehouse, database, schemas, file format, external stage, landing tables, and loads data from S3.
-- =================

-- Role and warehouse
CREATE ROLE IF NOT EXISTS EXAM_ROLE;
GRANT ROLE EXAM_ROLE TO USER VLADOKOEV;

CREATE WAREHOUSE IF NOT EXISTS COMPUTE_WH
    WAREHOUSE_SIZE = 'X-SMALL'
    AUTO_SUSPEND = 60
    AUTO_RESUME = TRUE;

USE WAREHOUSE COMPUTE_WH;

-- Database and schemas
CREATE DATABASE IF NOT EXISTS EXAM_PROJECT;
USE DATABASE EXAM_PROJECT;

CREATE SCHEMA IF NOT EXISTS STAGING; -- stage and file format only without data storage

CREATE SCHEMA IF NOT EXISTS CLEANSED; -- tables loaded from S3 (parquet format)
CREATE SCHEMA IF NOT EXISTS STAR; -- DIM and FACT tables (star schema)
CREATE SCHEMA IF NOT EXISTS PRESENTATION; -- materialized views for reporting

-- Grant role access
GRANT USAGE ON DATABASE EXAM_PROJECT TO ROLE EXAM_ROLE;
GRANT USAGE ON ALL SCHEMAS IN DATABASE EXAM_PROJECT TO ROLE EXAM_ROLE;
GRANT ALL PRIVILEGES ON ALL SCHEMAS IN DATABASE EXAM_PROJECT TO ROLE EXAM_ROLE;

-- Parquet format
CREATE OR REPLACE FILE FORMAT STAGING.PARQUET_FORMAT
    TYPE = PARQUET
    SNAPPY_COMPRESSION = TRUE
    USE_LOGICAL_TYPE = TRUE; -- load the datetime columns correctly and not as int

-- Create stage for loading from S3 processed folder
CREATE OR REPLACE STAGE STAGING.S3_PROCESSED_STAGE
    URL = 'URL_TO_S3'
    CREDENTIALS = (
        AWS_KEY_ID = 'AWS_KEY_ID'
        AWS_SECRET_KEY = 'AWS_SECRET_KEY'
    )
    FILE_FORMAT = STAGING.PARQUET_FORMAT;

-- Verify stage sees the Parquet files before loading
LIST @STAGING.S3_PROCESSED_STAGE;

-- Create landing tables (sales and product) in cleansed schema
CREATE OR REPLACE TABLE CLEANSED.SALES_CLEAN (
    SALES_ID INT,
    PRODUCT_ID INT,
    REGION VARCHAR,
    QUANTITY INT,
    PRICE NUMBER(10,2),
    SALE_DATE TIMESTAMP_NTZ,
    DISCOUNT NUMBER(5,4),
    ORDER_STATUS VARCHAR,
    DISCOUNTED_PRICE NUMBER(10,2),
    REVENUE NUMBER(10,2),
    PRICE_CATEGORY VARCHAR
);

CREATE OR REPLACE TABLE CLEANSED.PRODUCTS_CLEAN (
    PRODUCT_ID INT,
    CATEGORY VARCHAR,
    BRAND VARCHAR,
    RATING NUMBER(3,1),
    IN_STOCK BOOLEAN,
    LAUNCH_DATE TIMESTAMP_NTZ,
    DAYS_SINCE_LAUNCH INT
);

-- Load Parquet files from S3 into cleansed tables

TRUNCATE TABLE CLEANSED.SALES_CLEAN; -- remove records if data is reloaded again to avoid duplicating rows
COPY INTO CLEANSED.SALES_CLEAN
    FROM @STAGING.S3_PROCESSED_STAGE/sales_data.parquet
    FILE_FORMAT = STAGING.PARQUET_FORMAT
    MATCH_BY_COLUMN_NAME = CASE_INSENSITIVE
    ON_ERROR = 'CONTINUE';

TRUNCATE TABLE CLEANSED.PRODUCTS_CLEAN;
COPY INTO CLEANSED.PRODUCTS_CLEAN
    FROM @STAGING.S3_PROCESSED_STAGE/product_data.parquet
    FILE_FORMAT = STAGING.PARQUET_FORMAT
    MATCH_BY_COLUMN_NAME = CASE_INSENSITIVE
    ON_ERROR = 'CONTINUE';

-- Grant SELECT on loaded tables to EXAM_ROLE
GRANT SELECT ON ALL TABLES IN SCHEMA EXAM_PROJECT.CLEANSED TO ROLE EXAM_ROLE;

-- Quick validation on loaded sales and products data from S3.
SELECT 'SALES_CLEAN' AS table_name, COUNT(*) AS row_count FROM CLEANSED.SALES_CLEAN
UNION ALL
SELECT 'PRODUCTS_CLEAN' AS table_name, COUNT(*) AS row_count FROM CLEANSED.PRODUCTS_CLEAN;

SELECT * FROM CLEANSED.SALES_CLEAN LIMIT 5;

SELECT * FROM CLEANSED.PRODUCTS_CLEAN LIMIT 5;

-- =======================
-- BUILD THE STAR SCHEMA
-- DIM_DATE, DIM_PRODUCT, FACT_SALES from CLEANSED tables
-- =======================

-- DIM_DATE — one row per unique sale_date
-- DATE_KEY format YYYYMMDD is the standard key for dates
CREATE OR REPLACE TABLE STAR.DIM_DATE AS
SELECT DISTINCT
    TO_NUMBER(TO_CHAR(SALE_DATE::DATE, 'YYYYMMDD')) AS DATE_KEY,
    SALE_DATE::DATE AS FULL_DATE,
    YEAR(SALE_DATE) AS YEAR,
    MONTH(SALE_DATE) AS MONTH,
    DAY(SALE_DATE) AS DAY,
    QUARTER(SALE_DATE) AS QUARTER,
    DAYOFWEEK(SALE_DATE) AS DAY_OF_WEEK
FROM CLEANSED.SALES_CLEAN
ORDER BY FULL_DATE;

-- DIM_PRODUCT - product dimension from CLEANSED.PRODUCTS_CLEAN
CREATE OR REPLACE TABLE STAR.DIM_PRODUCT AS
SELECT
    PRODUCT_ID,
    CATEGORY,
    BRAND,
    RATING,
    IN_STOCK,
    LAUNCH_DATE::DATE AS LAUNCH_DATE,
    DAYS_SINCE_LAUNCH
FROM CLEANSED.PRODUCTS_CLEAN;

-- FACT_SALES - central fact table of sales and products
CREATE OR REPLACE TABLE STAR.FACT_SALES AS
SELECT
    s.SALES_ID AS SALE_ID,
    s.PRODUCT_ID,
    TO_NUMBER(TO_CHAR(s.SALE_DATE::DATE, 'YYYYMMDD')) AS DATE_KEY,
    s.SALE_DATE,
    p.CATEGORY,
    p.BRAND,
    p.RATING,
    CASE
        WHEN p.RATING >= 1 AND p.RATING < 2 THEN 'Very Low Rated Product'
        WHEN p.RATING >= 2 AND p.RATING < 3 THEN 'Low Rated Product'
        WHEN p.RATING >= 3 AND p.RATING < 4 THEN 'High Rated Product'
        WHEN p.RATING >= 4 AND p.RATING <= 5 THEN 'Very High Rated Product'
        ELSE 'No Rating'
    END AS RATING_CATEGORY,
    s.REGION,
    s.QUANTITY,
    s.PRICE,
    s.DISCOUNT,
    s.REVENUE,
    s.DISCOUNTED_PRICE,
    s.PRICE_CATEGORY,
    s.ORDER_STATUS
FROM CLEANSED.SALES_CLEAN s
LEFT JOIN CLEANSED.PRODUCTS_CLEAN p ON s.PRODUCT_ID = p.PRODUCT_ID;

-- Grant SELECT on star schema tables
GRANT SELECT ON ALL TABLES IN SCHEMA EXAM_PROJECT.STAR TO ROLE EXAM_ROLE;

-- Quick Validation steps
SELECT 'DIM_DATE'    AS table_name, COUNT(*) AS row_count FROM STAR.DIM_DATE
UNION ALL
SELECT 'DIM_PRODUCT' AS table_name, COUNT(*) AS row_count FROM STAR.DIM_PRODUCT
UNION ALL
SELECT 'FACT_SALES'  AS table_name, COUNT(*) AS row_count FROM STAR.FACT_SALES;

SELECT * FROM STAR.DIM_DATE;

SELECT * FROM STAR.DIM_PRODUCT;

SELECT * FROM STAR.FACT_SALES;

-- ==============================
-- MATERIALIZED ANALYTICAL VIEWS
-- ==============================

-- Revenue and order count grouped by region and calendar month
CREATE OR REPLACE MATERIALIZED VIEW PRESENTATION.MV_SALES_BY_REGION_MONTH AS
SELECT
    REGION,
    DATE_TRUNC('MONTH', SALE_DATE) AS SALES_MONTH,
    COUNT(*) AS ORDER_COUNT,
    SUM(REVENUE) AS TOTAL_REVENUE,
    ROUND(AVG(REVENUE),2) AS AVG_ORDER_REVENUE
FROM STAR.FACT_SALES
GROUP BY REGION, DATE_TRUNC('MONTH', SALE_DATE);

-- Revenue performance per product with category and brand context
CREATE OR REPLACE MATERIALIZED VIEW PRESENTATION.MV_TOP_PRODUCTS_BY_REVENUE AS
SELECT
    PRODUCT_ID,
    CATEGORY,
    BRAND,
    RATING_CATEGORY,
    COUNT(*) AS ORDER_COUNT,
    SUM(REVENUE) AS TOTAL_REVENUE,
    ROUND(AVG(REVENUE),2) AS AVG_REVENUE_PER_ORDER
FROM STAR.FACT_SALES
GROUP BY PRODUCT_ID, CATEGORY, BRAND, RATING_CATEGORY;

-- Monthly revenue trend in chronological order
CREATE OR REPLACE MATERIALIZED VIEW PRESENTATION.MV_REVENUE_TREND AS
SELECT
    DATE_TRUNC('MONTH', SALE_DATE)  AS SALES_MONTH,
    COUNT(*) AS ORDER_COUNT,
    SUM(REVENUE) AS TOTAL_REVENUE,
    ROUND(AVG(REVENUE),2) AS AVG_REVENUE
FROM STAR.FACT_SALES
GROUP BY DATE_TRUNC('MONTH', SALE_DATE);

-- Revenue and order performance broken down by price category
CREATE OR REPLACE MATERIALIZED VIEW PRESENTATION.MV_CATEGORY_PERFORMANCE AS
SELECT
    PRICE_CATEGORY,
    COUNT(*) AS ORDER_COUNT,
    SUM(REVENUE) AS TOTAL_REVENUE,
    ROUND(AVG(REVENUE),2) AS AVG_REVENUE,
    MIN(PRICE) AS MIN_PRICE,
    MAX(PRICE) AS MAX_PRICE
FROM STAR.FACT_SALES
GROUP BY PRICE_CATEGORY;

-- Grant SELECT on all views in PRESENTATION to EXAM_ROLE
GRANT SELECT ON ALL VIEWS IN SCHEMA EXAM_PROJECT.PRESENTATION TO ROLE EXAM_ROLE;

-- Final validation — all four MVs must return correct data
SELECT * FROM PRESENTATION.MV_SALES_BY_REGION_MONTH    LIMIT 5;
SELECT * FROM PRESENTATION.MV_TOP_PRODUCTS_BY_REVENUE  LIMIT 5;
SELECT * FROM PRESENTATION.MV_REVENUE_TREND            LIMIT 5;
SELECT * FROM PRESENTATION.MV_CATEGORY_PERFORMANCE     LIMIT 5;
